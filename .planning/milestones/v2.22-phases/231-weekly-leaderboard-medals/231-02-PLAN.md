# app/schemas/train.py
MEDAL_CLAIM_MAX_ITEMS: Final = 100   # about a year of both boards; caps the unclaimed read and the claim body

class UnclaimedMedal(BaseModel):          # ConfigDict(from_attributes=True)
    week_start: date                      # the Monday (UTC)
    board: LeaderboardBoardKind
    medal: MedalKind
    value: int                            # final points, or floored accuracy percent
    shared: bool                          # another row of the same week, board and medal exists

class UnclaimedMedalsResponse(BaseModel):
    medals: list[UnclaimedMedal]          # newest week first, then points before accuracy

class MedalKey(BaseModel):
    week_start: date
    board: LeaderboardBoardKind

class ClaimMedalsRequest(BaseModel):
    medals: list[MedalKey] = Field(min_length=1, max_length=MEDAL_CLAIM_MAX_ITEMS)

# GET  /api/train/medals/unclaimed -> UnclaimedMedalsResponse
# POST /api/train/medals/claim  body ClaimMedalsRequest -> 204 No Content

# app/repositories/train_medals_repository.py
@dataclass(frozen=True) class UnclaimedRow: week_start: date; board: LeaderboardBoardKind; medal: Medal; value: int; shared: bool
@dataclass(frozen=True) class TallyCount: board: LeaderboardBoardKind; user_id: int; medal: Medal; count: int
async def fetch_unclaimed(session, *, user_id: int, limit: int) -> list[UnclaimedRow]
async def mark_celebrated(session, *, user_id: int, keys: Sequence[tuple[date, LeaderboardBoardKind]], now_utc: datetime) -> int
async def fetch_medal_tallies(session, *, user_ids: Sequence[int]) -> list[TallyCount]

# app/services/train_medals.py

async def get_unclaimed_medals(session, *, user_id: int) -> list[UnclaimedMedalItem]

# app/services/train_leaderboard.py
def visible_keys(kind: LeaderboardBoardKind, aggregates: Sequence[WeeklyAggregate], *, viewer_id: int) -> list[int]
