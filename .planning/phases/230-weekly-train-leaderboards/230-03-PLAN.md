Put the weekly leaderboard on the Train landing: one "This week" card with Points and Accuracy tabs, the deadline countdown, the viewer's highlighted row with neighbours, the pass-target / qualifier / not-entered hints, the private hidden and guest rows, and the guest sign-up nudge, fed by the Plan 01 endpoint.

Purpose: this is where users see the competition every time they open Train (D-07, D-08). The card is the main consumer of the API contract, so it lands right after the backend foundation, in the same wave as the backend session-delta plan (230-02), with which it shares no file. The card never reads rank_without_session; the score-screen plan (230-05) does.

```typescript
export type LeaderboardBoardKind = 'points' | 'accuracy';
export type LeaderboardVisibility = 'public' | 'hidden' | 'guest';
export interface LeaderboardRow {
  rank: number;
  name: string;
  value: number;          // points, or floored accuracy percent
  puzzles: number;        // all solves (points) or non-filler solves (accuracy)
  tentative: boolean;     // accuracy only
  is_viewer: boolean;
  visibility: LeaderboardVisibility; // 'public' except the viewer's own hidden/guest row
  gap_before: boolean;
}
export interface LeaderboardViewer {
  rank: number;
  rank_without_session: number | null;
  tentative: boolean;
  puzzles_to_qualify: number;
  visibility: LeaderboardVisibility;
}
export interface LeaderboardPassTarget { name: string; points_needed: number; }
export interface LeaderboardBoard {
  rows: LeaderboardRow[];
  viewer: LeaderboardViewer | null;
  pass_target: LeaderboardPassTarget | null; // points board only
}
export interface TrainLeaderboardResponse {
  week_start: string;
  week_end: string;
  seconds_remaining: number;
  points: LeaderboardBoard;
  accuracy: LeaderboardBoard;
}
```
