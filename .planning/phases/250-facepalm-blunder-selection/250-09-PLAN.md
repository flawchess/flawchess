```sql
-- O1 coverage after the spread pass (user-side blunders of base-stamped games)
SELECT count(*) AS user_blunders,
       count(f.maia_reply_spread) AS with_spread,
       round(100.0 * count(f.maia_reply_spread) / nullif(count(*), 0), 3) AS pct_with_spread,
       count(f.facepalm_score) AS with_score,
       round(100.0 * count(f.facepalm_score) / nullif(count(*), 0), 3) AS pct_with_score,
       count(*) FILTER (WHERE f.facepalm_score >= 0.7) AS qualifying,
       count(*) FILTER (WHERE f.maia_reply_spread IS NOT NULL AND f.facepalm_score IS NULL AND gp.pv IS NULL) AS null_no_line,
       count(*) FILTER (WHERE f.maia_reply_spread IS NOT NULL AND f.facepalm_score IS NULL AND gp.pv IS NOT NULL) AS null_short_line_or_no_played_prob,
       count(*) FILTER (WHERE f.maia_reply_spread IS NULL) AS null_no_spread,
       count(*) FILTER (WHERE g.facepalm_spread_completed_at IS NULL) AS in_unstamped_games
FROM game_flaws f
JOIN games g ON g.id = f.game_id
LEFT JOIN game_positions gp ON gp.game_id = f.game_id AND gp.user_id = f.user_id AND gp.ply = f.ply + 1
WHERE f.severity = 2
  AND (f.ply % 2 = 0) = (g.user_color = 'white')
  AND g.maia_flaws_completed_at IS NOT NULL;

-- O2 slice check: only user blunders got values in the slice
SELECT count(*) FILTER (WHERE f.maia_reply_spread IS NOT NULL) AS spread_rows,
       count(*) FILTER (WHERE (f.maia_reply_spread IS NOT NULL OR f.facepalm_score IS NOT NULL)
                         AND NOT (f.severity = 2 AND (f.ply % 2 = 0) = (g.user_color = 'white'))) AS out_of_scope_rows
FROM game_flaws f JOIN games g ON g.id = f.game_id
WHERE g.facepalm_spread_completed_at >= '<slice start UTC>';

-- O3 hook live after the deploy
SELECT count(*) AS scored_games,
       count(*) FILTER (WHERE g.facepalm_spread_completed_at IS NOT NULL) AS with_facepalm_stamp
FROM games g WHERE g.maia_flaws_completed_at >= '<deploy completed UTC>';

-- O4 connections (before launch and while running)
SELECT count(*) FROM pg_stat_activity;

-- O5 schema head
SELECT version_num FROM alembic_version;
```
